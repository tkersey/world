+fn allocationBytes(comptime T: type, value: T) usize {
+    var size: usize = 0;
+    switch (@typeInfo(T)) {
+        .pointer => |info| {
+            size = value.len *| @sizeOf(info.child);
+            for (value) |item| size +|= allocationBytes(info.child, item);
+        },
+        .optional => |info| if (value) |item| {
+            size = allocationBytes(info.child, item);
+        },
+        .@"struct" => |info| inline for (info.fields) |field| {
+            size +|= allocationBytes(field.type, @field(value, field.name));
+        },
+        .@"union" => |info| inline for (info.fields) |field| {
+            if (std.mem.eql(u8, @tagName(value), field.name))
+                return allocationBytes(field.type, @field(value, field.name));
+        },
+        else => {},
+    }
